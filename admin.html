const bstr = atob(arr[1]); let n = bstr.length; const u8arr = new Uint8Array(n); while (n--) { u8arr[n] = bstr.charCodeAt(n); } return new Blob([u8arr], { type: mime }); } /** * Sube la imagen comprimida a ImgBB usando FormData nativo */ async function uploadToImgBB(base64Data) { if (!IMGBB_API_KEY || IMGBB_API_KEY.includes("COLOCA_AQUI")) { throw new Error("Debes colocar una API Key válida de ImgBB en el código.");
